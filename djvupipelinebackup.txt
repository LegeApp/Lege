// Thread-local memory pools for DJVU pipeline to reduce allocations
thread_local! {
    static DJVU_GRAYSCALE_BUFFER: RefCell<Vec<u8>> = RefCell::new(Vec::with_capacity(10_000_000));
    static DJVU_BINARIZED_BUFFER: RefCell<Vec<u8>> = RefCell::new(Vec::with_capacity(10_000_000));
    static DJVU_TEMP_BUFFER: RefCell<Vec<u8>> = RefCell::new(Vec::with_capacity(5_000_000));
    static DJVU_INFERENCE_BUFFER: RefCell<Vec<u8>> = RefCell::new(Vec::with_capacity(640 * 640 * 3));
    // Engine pool with multiple pre-warmed engines
    static DJVU_ENGINE_POOL: RefCell<VecDeque<Arc<tokio::sync::Mutex<crate::engine::PaddleXEngine>>>> = RefCell::new(VecDeque::new());
    // Thread-local binarizer for reuse
    static HEAVY_SAUVOLA_PROCESSOR: RefCell<Option<HeavySauvolaProcessor>> = RefCell::new(None);
}

static DJVU_STANDARD_WIDTH: AtomicU32 = AtomicU32::new(0);
static DJVU_STANDARD_HEIGHT: AtomicU32 = AtomicU32::new(0);

pub(crate) fn reset_standard_dimensions() {
    DJVU_STANDARD_WIDTH.store(0, Ordering::Relaxed);
    DJVU_STANDARD_HEIGHT.store(0, Ordering::Relaxed);
}

/// Get or create a warmed inference engine for this worker thread
fn get_or_create_warmed_engine(config: &crate::engine::PaddleXConfig) -> Result<Arc<tokio::sync::Mutex<crate::engine::PaddleXEngine>>> {
    DJVU_ENGINE_POOL.with(|pool| {
        let mut pool_ref = pool.borrow_mut();
        
        // Try to get an available engine from the pool
        if let Some(engine) = pool_ref.pop_front() {
            crate::debug_log!("DJVU: Reusing warmed engine from pool");
            Ok(engine)
        } else {
            crate::debug_log!("DJVU: Creating new engine for pool");
            // Create a new engine with the provided config
            let model_path = crate::runtime_asset_path("paddle-layout.onnx")
                .to_string_lossy()
                .to_string();
            match crate::engine::PaddleXEngine::new(&model_path, config.clone()) {
                Ok(engine) => {
                    let wrapped_engine = Arc::new(tokio::sync::Mutex::new(engine));
                    Ok(wrapped_engine)
                }
                Err(err) => Err(err)
            }
        }
    })
}

/// Return an engine to the pool for reuse (thread-local)
fn return_engine_to_pool(engine: Arc<tokio::sync::Mutex<crate::engine::PaddleXEngine>>) {
    DJVU_ENGINE_POOL.with(|pool| {
        let mut pool_ref = pool.borrow_mut();
        // Limit pool size to prevent excessive memory usage
        if pool_ref.len() < 2 { // Keep max 2 engines per thread
            pool_ref.push_back(engine);
            crate::debug_log!("DJVU: Returned engine to pool (pool size: {})", pool_ref.len());
        }
    });
}

/// Warm up the inference engine with a dummy inference to reduce first-batch latency
async fn warm_engine_if_needed(engine: &Arc<tokio::sync::Mutex<crate::engine::PaddleXEngine>>) -> Result<()> {
    let mut engine_guard = engine.lock().await;
    
    // Create a small dummy image for warming (64x64 RGB)
    let dummy_image = image::RgbImage::new(64, 64);
    
    crate::debug_log!("DJVU: Warming engine with dummy inference");
    let warm_start = std::time::Instant::now();
    
    match engine_guard.detect_single_async(&dummy_image).await {
        Ok(_) => {
            crate::debug_log!("DJVU: Engine warmed successfully in {:?}", warm_start.elapsed());
            Ok(())
        }
        Err(err) => {
            crate::warn!("DJVU: Engine warming failed: {}", err);
            // Don't fail the pipeline if warming fails
            Ok(())
        }
    }
}



/// Use pooled buffer for temporary operations
fn with_djvu_temp_buffer<F, R>(min_size: usize, f: F) -> R
where
    F: FnOnce(&mut Vec<u8>) -> R,
{
    DJVU_TEMP_BUFFER.with(|buffer| {
        let mut buf = buffer.borrow_mut();
        let current_capacity = buf.capacity();
        if current_capacity < min_size {
            buf.reserve(min_size - current_capacity);
        }
        buf.clear();
        f(&mut buf)
    })
}

/// Use pooled buffer for binarized data
fn with_djvu_binarized_buffer<F, R>(min_size: usize, f: F) -> R
where
    F: FnOnce(&mut Vec<u8>) -> R,
{
    DJVU_BINARIZED_BUFFER.with(|buffer| {
        let mut buf = buffer.borrow_mut();
        let current_capacity = buf.capacity();
        if current_capacity < min_size {
            buf.reserve(min_size - current_capacity);
        }
        buf.clear();
        f(&mut buf)
    })
}

/// Prefetch and cache the next N images from the input channel to reduce pipeline stalls
async fn prefetch_pages(
    in_channels: &mut TypedInChannels<DjvuInferenceResult>,
    prefetch_count: usize,
) -> Result<Vec<Arc<DjvuInferenceResult>>, ()> {
    let mut prefetched = Vec::with_capacity(prefetch_count);
    
    // Try to get up to prefetch_count pages without blocking
    for _ in 0..prefetch_count {
        match in_channels.recv_any().await {
            Ok((_, Some(page))) => {
                prefetched.push(page);
            }
            Ok((_, None)) => {
                // Channel closed - return what we have
                break;
            }
            Err(_) => {
                // Error receiving - return what we have
                break;
            }
        }
    }
    
    Ok(prefetched)
}

/// DJVU Batch Accumulator - accumulates pages for batch processing
pub struct DjvuBatchAccumulator {
    pub config: Arc<PipelineConfig>,
    pub total_pages: usize,
}

#[async_trait]
impl TypedAction for DjvuBatchAccumulator {
    type I = RenderedPageData;
    type O = DjvuBatchRenderedData;

    async fn run(
        &self,
        mut in_channels: TypedInChannels<Self::I>,
        out_channels: TypedOutChannels<Self::O>,
        env: Arc<EnvVar>,
    ) -> Output {
        let batch_size = self
            .config
            .max_inference_batch_size()
            .map(|m| std::cmp::min(m, self.config.batch_size()))
            .unwrap_or_else(|| self.config.batch_size());
        let min_pages_for_batching = self.config.min_pages_for_batching();
        let timeout_ms = self.config.batch_timeout_ms();
        let initial_single_pages = self.config.initial_single_pages(); // Number of single pages before batching

        crate::info_log!(
            "DjvuBatchAccumulator: start • total_pages={} • batch_size={} • min_pages_for_batching={} • timeout_ms={} • initial_single={} ",
            self.total_pages,
            batch_size,
            min_pages_for_batching,
            timeout_ms,
            initial_single_pages
        );

        // Adaptive batching: send initial pages as singles to warm up workers, then batch
        let mut pages_received = 0;
        let mut current_batch_pages = Vec::with_capacity(batch_size);
        let mut current_batch_indices = Vec::with_capacity(batch_size);
        
        // Pre-allocate vector to reduce reallocations
        let mut single_batch = Vec::with_capacity(1);
        let mut single_indices = Vec::with_capacity(1);
        
        crate::info_log!(
            "DjvuBatchAccumulator: Adaptive mode • initial_single={} • batch_size={} • timeout_ms={} • total_pages={}",
            initial_single_pages,
            batch_size,
            timeout_ms,
            self.total_pages
        );

        loop {
            match tokio::time::timeout(std::time::Duration::from_millis(timeout_ms), in_channels.recv_any()).await {
                Ok(Ok((_sender_id, rendered_data_opt))) => {
                    let rendered_data = match rendered_data_opt {
                        Some(d) => d,
                        None => {
                            // Flush any remaining pages in the final batch
                            if !current_batch_pages.is_empty() {
                                crate::info_log!(
                                    "DjvuBatchAccumulator: flushing final batch • size={}",
                                    current_batch_pages.len()
                                );
                                let batch = DjvuBatchRenderedData {
                                    pages: std::mem::take(&mut current_batch_pages),
                                    page_indices: std::mem::take(&mut current_batch_indices),
                                };
                                out_channels.broadcast(batch).await;
                            }
                            crate::info_log!(
                                "DjvuBatchAccumulator: input closed • pages_received={}/{}",
                                pages_received,
                                self.total_pages
                            );
                            break;
                        }
                    };

                    pages_received += 1;

                    // Adaptive batching: send first N pages as singles, then batch
                    if pages_received <= initial_single_pages {
                        // Send as individual batch for faster startup
                        single_batch.clear();
                        single_indices.clear();
                        single_batch.push((*rendered_data).clone());
                        single_indices.push(rendered_data.index);
                        
                        let single_batch_data = DjvuBatchRenderedData {
                            pages: single_batch.clone(),
                            page_indices: single_indices.clone(),
                        };
                        out_channels.broadcast(single_batch_data).await;
                        crate::debug_log!(
                            "DjvuBatchAccumulator: → adaptive single • page={} • received={}/{}",
                            rendered_data.index,
                            pages_received,
                            initial_single_pages
                        );
                    } else {
                        // Add page to current batch for throughput optimization
                        current_batch_pages.push((*rendered_data).clone());
                        current_batch_indices.push(rendered_data.index);

                        // If batch is full, send it and start a new one
                        if current_batch_pages.len() >= batch_size {
                            crate::debug_log!(
                                "DjvuBatchAccumulator: → adaptive batch full • size={}",
                                current_batch_pages.len()
                            );
                            let batch = DjvuBatchRenderedData {
                                pages: std::mem::take(&mut current_batch_pages),
                                page_indices: std::mem::take(&mut current_batch_indices),
                            };
                            out_channels.broadcast(batch).await;
                        }
                    }
                }
                Ok(Err(_e)) => {
                    crate::warn_log!("DjvuBatchAccumulator: recv error while waiting for page in batch mode");
                    break;
                }
                Err(_elapsed) => {
                    // Timeout: flush partial batch if any (but not if we're still in single page mode)
                    if pages_received > initial_single_pages && !current_batch_pages.is_empty() {
                        crate::info_log!(
                            "DjvuBatchAccumulator: timeout flush • size={} • received={}/{}",
                            current_batch_pages.len(),
                            pages_received,
                            self.total_pages
                        );
                        let batch = DjvuBatchRenderedData {
                            pages: std::mem::take(&mut current_batch_pages),
                            page_indices: std::mem::take(&mut current_batch_indices),
                        };
                        out_channels.broadcast(batch).await;
                    } else if pages_received <= initial_single_pages {
                        // Still in single page mode - continue waiting for more pages
                        crate::info_log!(
                            "DjvuBatchAccumulator: timeout in single mode • received={}/{}",
                            pages_received,
                            initial_single_pages
                        );
                    } else {
                        crate::info_log!(
                            "DjvuBatchAccumulator: timeout • no pages for {}ms • received={}/{}",
                            timeout_ms,
                            pages_received,
                            self.total_pages
                        );
                    }
                    // continue loop waiting again
                }
            }
        }

        let mut out_channels = out_channels;
        for id in out_channels.get_receiver_ids() {
            out_channels.close(&id);
        }
        Output::Out(None)
    }
}

/// DJVU Batch Layout Detection Action - performs batch inference on groups of pages
/// Performs batch layout detection on DJVU pages using the inference engine.
/// Processes pages in batches for optimal GPU utilization and performance.
pub struct DjvuBatchLayoutDetectionAction {
    pub config: Arc<PipelineConfig>,
}

#[async_trait]
impl TypedAction for DjvuBatchLayoutDetectionAction {
    type I = DjvuBatchRenderedData;
    type O = DjvuInferenceResult;

    async fn run(
        &self,
        mut in_channels: TypedInChannels<Self::I>,
        mut out_channels: TypedOutChannels<Self::O>,
        env: Arc<EnvVar>,
    ) -> Output {
        let inference_callback = env.get::<Arc<dyn Fn(usize, usize) + Send + Sync + 'static>>("inference_callback");
        let mut inferred_count = 0usize;
        let total_pages = env.get::<usize>("total_pages").unwrap_or(1);

        // Create engine config for this worker
        let engine_config = crate::engine::PaddleXConfig {
            confidence_threshold: self.config.confidence_threshold,
            nms_threshold: self.config.nms_threshold,
            iou_threshold: self.config.nms_threshold,
            batch_size: self.config.batch_size().min(8), // Cap batch size for DJVU
        };

        // Create a dedicated engine instance for this worker (same pattern as PDF pipeline)
        let engine = if self.config.enable_layout_detection() {
            match get_or_create_warmed_engine(&engine_config) {
                Ok(engine) => {
                    // Warm the engine on first use to reduce first-batch latency
                    if let Err(err) = warm_engine_if_needed(&engine).await {
                        crate::warn!("DjvuBatchLayoutDetection: Engine warming failed: {}", err);
                    }
                    Some(engine)
                }
                Err(err) => {
                    crate::warn!("DjvuBatchLayoutDetection: Failed to create inference engine: {}", err);
                    None
                }
            }
        } else {
            None
        };

        while let Ok((_, batch_opt)) = in_channels.recv_any().await {
            let batch = match batch_opt {
                Some(b) => b,
                None => {
                    crate::info_log!("DjvuBatchLayoutDetection: Upstream channel closed; stopping.");
                    break;
                }
            };

            let batch_len = batch.pages.len();
            let first_idx = batch.page_indices.first().copied();
            let last_idx = batch.page_indices.last().copied();
            crate::info_log!(
                "DjvuBatchLayoutDetection: received batch • size={} • indices={:?}→{:?}",
                batch_len,
                first_idx,
                last_idx
            );

            let batch_start = std::time::Instant::now();
            let mut processed_pages = 0usize;

            // Use batch inference for better performance, but with improved error handling
            if batch_len > 1 && self.config.enable_layout_detection() && engine.is_some() {
                // Extract inference images for batch processing
                let inference_images: Vec<_> = batch.pages.iter()
                    .map(|page| (*page.inference_image).clone())
                    .collect();

                // Run batch inference with the engine
                match {
                    let engine_guard = engine.as_ref().unwrap().clone();
                    let images = inference_images.clone();
                    let indices = batch.page_indices.clone();
                    tokio::spawn(async move {
                        let mut guard = engine_guard.lock().await;
                        guard.detect_batch_with_indices_async(&images, &indices).await
                    }).await
                } {
                    Ok(Ok(batch_detections)) => {
                        // Process each page with its detections
                        for (i, page_data) in batch.pages.iter().enumerate() {
                            let page_index = page_data.index;
                            let mut detections = batch_detections.get(i).cloned().unwrap_or_default();
                            
                            // Filter out title detections like in PDF pipeline
                            detections.retain(|d| {
                                if let Some(name) = &d.class_name {
                                    if name.to_ascii_lowercase().contains("title") {
                                        crate::debug_log!("DjvuBatchLayoutDetection: Page {} - skipping title detection", page_index);
                                        return false;
                                    }
                                }
                                true
                            });

                            // Convert detections to page-space coordinates
                            let page_w_u32 = page_data.high_res_image.width();
                            let page_h_u32 = page_data.high_res_image.height();
                            if !detections.is_empty() {
                                let spec = InferenceResizeSpec {
                                    target: self.config.inference_size(),
                                    ..Default::default()
                                };
                                for det in &mut detections {
                                    let orig = det.bbox;
                                    let remapped = maybe_remap_bbox_from_infer(orig, page_w_u32, page_h_u32, &spec);
                                    if remapped != orig {
                                        det.bbox = remapped;
                                        crate::dbglog!(
                                            "[djvu-batch] remapped bbox page={} {:?} -> {:?}",
                                            page_index, orig, remapped
                                        );
                                    }
                                }
                            }

                            // Create binarized image (will be computed in the page processor)
                            let binarized = Vec::new(); 

                            // Distribute pages to downstream processors (round-robin) to avoid duplicate work
                            let receivers = out_channels.get_receiver_ids();
                            if receivers.is_empty() {
                                // No receivers; safe fallback to broadcast
                                out_channels.broadcast(DjvuInferenceResult {
                                    index: page_index,
                                    high_res_image: page_data.high_res_image.clone(),
                                    binarized,
                                    detections,
                                }).await;
                            } else {
                                // Round-robin index carried via a static to persist across batches
                                use std::sync::atomic::{AtomicUsize, Ordering as Ord2};
                                static RR_IDX: AtomicUsize = AtomicUsize::new(0);
                                let idx = RR_IDX.fetch_add(1, Ord2::Relaxed) % receivers.len();
                                let mut sent = false;
                                for offset in 0..receivers.len() {
                                    let target = &receivers[(idx + offset) % receivers.len()];
                                    let res = out_channels
                                        .send_to(target, DjvuInferenceResult {
                                            index: page_index,
                                            high_res_image: page_data.high_res_image.clone(),
                                            binarized: binarized.clone(),
                                            detections: detections.clone(),
                                        })
                                        .await;
                                    if res.is_ok() { sent = true; break; }
                                }
                                if !sent {
                                    crate::dbglog!(
                                        "[djvu-batch] Failed to send page {} to any receiver; falling back to broadcast",
                                        page_index
                                    );
                                    out_channels
                                        .broadcast(DjvuInferenceResult {
                                            index: page_index,
                                            high_res_image: page_data.high_res_image.clone(),
                                            binarized,
                                            detections,
                                        })
                                        .await;
                                }
                            }

                            processed_pages += 1;
                            inferred_count += 1;
                            if let Some(callback) = &inference_callback {
                                callback(inferred_count, total_pages);
                            }
                        }
                    }
                    Ok(Err(err)) => {
                        crate::error!("DjvuBatchLayoutDetection: Batch inference failed: {}", err);
                        // Fall back to individual processing with better error handling
                        for page_data in batch.pages.iter() {
                            let page_index = page_data.index;
                            let mut detections = Vec::new();
                            
                            // Process individually with available engine
                            if self.config.enable_layout_detection() && engine.is_some() {
                                match {
                                    let engine_guard = engine.as_ref().unwrap().clone();
                                    let img = page_data.inference_image.clone();
                                    tokio::spawn(async move {
                                        let mut guard = engine_guard.lock().await;
                                        guard.detect_single_async(img.as_ref()).await
                                    }).await
                                } {
                                    Ok(Ok(mut dets)) => {
                                        dets.retain(|d| {
                                            if let Some(name) = &d.class_name {
                                                if name.to_ascii_lowercase().contains("title") {
                                                    crate::debug_log!("DjvuBatchLayoutDetection: Page {} - skipping title detection", page_index);
                                                    return false;
                                                }
                                            }
                                            true
                                        });
                                        detections = dets;
                                    }
                                    Ok(Err(err)) => {
                                        crate::error!("DjvuBatchLayoutDetection: Single inference failed for page {}: {}", page_index, err);
                                    }
                                    Err(err) => {
                                        crate::error!("DjvuBatchLayoutDetection: Task spawn failed for page {}: {}", page_index, err);
                                    }
                                }
                            }
                            
                            let binarized = Vec::new();

                            // Distribute to a single processor to avoid duplicate work
                            let receivers = out_channels.get_receiver_ids();
                            if receivers.is_empty() {
                                out_channels.broadcast(DjvuInferenceResult {
                                    index: page_index,
                                    high_res_image: page_data.high_res_image.clone(),
                                    binarized,
                                    detections,
                                }).await;
                            } else {
                                use std::sync::atomic::{AtomicUsize, Ordering as Ord2};
                                static RR_IDX: AtomicUsize = AtomicUsize::new(0);
                                let idx = RR_IDX.fetch_add(1, Ord2::Relaxed) % receivers.len();
                                let mut sent = false;
                                for offset in 0..receivers.len() {
                                    let target = &receivers[(idx + offset) % receivers.len()];
                                    let res = out_channels
                                        .send_to(target, DjvuInferenceResult {
                                            index: page_index,
                                            high_res_image: page_data.high_res_image.clone(),
                                            binarized: binarized.clone(),
                                            detections: detections.clone(),
                                        })
                                        .await;
                                    if res.is_ok() { sent = true; break; }
                                }
                                if !sent {
                                    crate::dbglog!(
                                        "[djvu-batch-fallback] Failed to send page {} to any receiver; falling back to broadcast",
                                        page_index
                                    );
                                    out_channels
                                        .broadcast(DjvuInferenceResult {
                                            index: page_index,
                                            high_res_image: page_data.high_res_image.clone(),
                                            binarized,
                                            detections,
                                        })
                                        .await;
                                }
                            }

                            processed_pages += 1;
                            inferred_count += 1;
                            if let Some(callback) = &inference_callback {
                                callback(inferred_count, total_pages);
                            }
                        }
                    }
                    Err(join_err) => {
                        crate::error!("DjvuBatchLayoutDetection: Task join failed: {}", join_err);
                        // Fall back to individual processing without engine
                        for page_data in batch.pages.iter() {
                            let page_index = page_data.index;
                            let detections = Vec::new();
                            let binarized = Vec::new();

                            out_channels.broadcast(DjvuInferenceResult {
                                index: page_index,
                                high_res_image: page_data.high_res_image.clone(),
                                binarized,
                                detections,
                            }).await;

                            processed_pages += 1;
                            inferred_count += 1;
                            if let Some(callback) = &inference_callback {
                                callback(inferred_count, total_pages);
                            }
                        }
                    }
                }
            } else {
                // Process pages individually (fallback for single pages or when batch inference is disabled)
                for page_data in batch.pages.iter() {
                    let page_index = page_data.index;
                    let is_cover_page = crate::should_treat_as_cover_page(page_index, &self.config);

                    let mut detections = if !self.config.enable_layout_detection() || is_cover_page {
                        if is_cover_page {
                            let (w, h) = (
                                page_data.high_res_image.width() as f32,
                                page_data.high_res_image.height() as f32,
                            );
                            vec![crate::engine::Detection {
                                class_id: 1,
                                class_name: Some("cover_page".to_string()),
                                confidence: 1.0,
                                bbox: [0.0, 0.0, w, h],
                                context: None,
                            }]
                        } else {
                            Vec::new()
                        }
                    } else if let Some(engine) = &engine {
                        match {
                            let engine_guard = engine.clone();
                            let img = page_data.inference_image.clone();
                            tokio::spawn(async move {
                                let mut guard = engine_guard.lock().await;
                                guard.detect_single_async(img.as_ref()).await
                            }).await
                        } {
                            Ok(Ok(mut dets)) => {
                                dets.retain(|d| {
                                    if let Some(name) = &d.class_name {
                                        if name.to_ascii_lowercase().contains("title") {
                                            crate::debug_log!("DjvuBatchLayoutDetection: Page {} - skipping title detection", page_index);
                                            return false;
                                        }
                                    }
                                    true
                                });
                                dets
                            }
                            Ok(Err(err)) => {
                                crate::error!("DjvuBatchLayoutDetection: Inference failed for page {}: {}", page_index, err);
                                Vec::new()
                            }
                            Err(join_err) => {
                                crate::error!("DjvuBatchLayoutDetection: Task join failed for page {}: {}", page_index, join_err);
                                Vec::new()
                            }
                        }
                    } else {
                        crate::warn!(
                            "DjvuBatchLayoutDetection: Layout detection disabled but engine unavailable for page {}; returning empty detections",
                            page_index
                        );
                        Vec::new()
                    };

                    // Convert detections to page-space coordinates
                    let page_w_u32 = page_data.high_res_image.width();
                    let page_h_u32 = page_data.high_res_image.height();
                    if !detections.is_empty() {
                        let spec = InferenceResizeSpec {
                            target: self.config.inference_size(),
                            ..Default::default()
                        };
                        for det in &mut detections {
                            let orig = det.bbox;
                            let remapped = maybe_remap_bbox_from_infer(orig, page_w_u32, page_h_u32, &spec);
                            if remapped != orig {
                                det.bbox = remapped;
                            }
                        }
                    }

                    // Create binarized image (placeholder for now - will be computed in page processor)
                    let binarized = Vec::new();

                    // Distribute to a single processor to avoid duplicate work
                    let receivers = out_channels.get_receiver_ids();
                    if receivers.is_empty() {
                        out_channels.broadcast(DjvuInferenceResult {
                            index: page_index,
                            high_res_image: page_data.high_res_image.clone(),
                            binarized,
                            detections,
                        }).await;
                    } else {
                        use std::sync::atomic::{AtomicUsize, Ordering as Ord2};
                        static RR_IDX: AtomicUsize = AtomicUsize::new(0);
                        let idx = RR_IDX.fetch_add(1, Ord2::Relaxed) % receivers.len();
                        let mut sent = false;
                        for offset in 0..receivers.len() {
                            let target = &receivers[(idx + offset) % receivers.len()];
                            let res = out_channels
                                .send_to(target, DjvuInferenceResult {
                                    index: page_index,
                                    high_res_image: page_data.high_res_image.clone(),
                                    binarized: binarized.clone(),
                                    detections: detections.clone(),
                                })
                                .await;
                            if res.is_ok() { sent = true; break; }
                        }
                        if !sent {
                            crate::dbglog!(
                                "[djvu-individual] Failed to send page {} to any receiver; falling back to broadcast",
                                page_index
                            );
                            out_channels
                                .broadcast(DjvuInferenceResult {
                                    index: page_index,
                                    high_res_image: page_data.high_res_image.clone(),
                                    binarized,
                                    detections,
                                })
                                .await;
                        }
                    }

                    processed_pages += 1;
                    inferred_count += 1;
                    if let Some(callback) = &inference_callback {
                        callback(inferred_count, total_pages);
                    }
                }
            }

            let elapsed = batch_start.elapsed();
            crate::info_log!(
                "DjvuBatchLayoutDetection: batch complete • size={} • processed={} • elapsed={:.2}s • {:.2} pages/sec",
                batch_len,
                processed_pages,
                elapsed.as_secs_f64(),
                processed_pages as f64 / elapsed.as_secs_f64()
            );
        }

        let mut out_channels = out_channels;
        for id in out_channels.get_receiver_ids() {
            out_channels.close(&id);
        }
        Output::Out(None)
    }
}

/// DJVU Page Processor Action
/// Processes a single page: binarizes, runs layout detection, exports data
/// Now uses pre-computed inference results from batch processing
pub struct DjvuPageProcessorAction {
    pub config: Arc<PipelineConfig>,
    pub orchestrator_state: Arc<DjvuOrchestratorState>,
}

#[async_trait]
impl TypedAction for DjvuPageProcessorAction {
    type I = DjvuInferenceResult; // Now takes inference results from batch processing
    type O = usize; // Returns page index when complete

    async fn run(
        &self,
        mut in_channels: TypedInChannels<Self::I>,
        out_channels: TypedOutChannels<Self::O>,
        env: Arc<EnvVar>,
    ) -> Output {
        // Lazily initialize a heavy-duty processor once per worker if requested
        if self.config.binarization.use_heavy_duty {
            HEAVY_SAUVOLA_PROCESSOR.with(|proc_cell| {
                if proc_cell.borrow().is_none() {
                    match HeavySauvolaProcessor::new() {
                        Ok(p) => *proc_cell.borrow_mut() = Some(p),
                        Err(e) => {
                            #[cfg(feature = "debug-logging")]
                            crate::debug_log!("[djvu-page] Heavy-duty init failed; will fallback: {}", e);
                        }
                    }
                }
            });
        }

        // Create a buffer pool for image processing
        let mut reusable_image_buffer: Vec<u8> = Vec::with_capacity(640 * 640 * 3); // Typical max size

        while let Ok((_sender_id, inference_result_opt)) = in_channels.recv_any().await {
            let inference_result = match inference_result_opt {
                Some(d) => d,
                None => break,
            };

            let page_index = inference_result.index;
            dbglog!("[djvu-page] start page {}", page_index);

            // 1. Use pre-computed detections from batch inference
            let mut detections = inference_result.detections.clone();

            // High-res image clone for potential margin adjustments
            // Important: Do NOT invert the RGB here. For inverted input, we only pre-invert
            // during grayscale binarization. Keeping RGB untouched preserves correct colors
            // for DjVu BG44 image regions.
            
            // For now, clone the image directly - buffer reuse can be optimized later
            let mut adjusted_image: ImageBuffer<Rgb<u8>, Vec<u8>> = (*inference_result.high_res_image).clone();

            let mut page_w = adjusted_image.width() as usize;
            let mut page_h = adjusted_image.height() as usize;
            let mut page_w_u32 = adjusted_image.width();
            let mut page_h_u32 = adjusted_image.height();

            // Capture standard dimensions for margin processing (first page wins)
            if DJVU_STANDARD_WIDTH.load(Ordering::Relaxed) == 0
                || DJVU_STANDARD_HEIGHT.load(Ordering::Relaxed) == 0
            {
                DJVU_STANDARD_WIDTH.store(page_w_u32, Ordering::Relaxed);
                DJVU_STANDARD_HEIGHT.store(page_h_u32, Ordering::Relaxed);
            }

            // 2. Convert detections to page-space if the engine left them in inference-space coordinates.
            if !detections.is_empty() {
                let spec = InferenceResizeSpec {
                    target: self.config.inference_size(),
                    ..Default::default()
                };
                for det in &mut detections {
                    let orig = det.bbox;
                    let remapped = maybe_remap_bbox_from_infer(orig, page_w_u32, page_h_u32, &spec);
                    if remapped != orig {
                        dbglog!(
                            "[djvu-page] remapped bbox page={} {:?} -> {:?}",
                            page_index, orig, remapped
                        );
                        det.bbox = remapped;
                    } else {
                        dbglog!(
                            "[djvu-page] bbox already page-space page={} {:?}",
                            page_index, orig
                        );
                    }
                }
            }

            // 3. Apply margin processing if enabled (pre-binarization)
            let margin_setting = self.config.margin_settings();
            if margin_setting != MarginSettings::None && !detections.is_empty() {
                let filter_for_cropping = matches!(margin_setting, MarginSettings::CropAndResize);
                if let Some(bounds) = margin::calculate_content_bounds(
                    &detections,
                    page_w_u32,
                    page_h_u32,
                    filter_for_cropping,
                ) {
                    let mut std_width = DJVU_STANDARD_WIDTH.load(Ordering::Relaxed);
                    let mut std_height = DJVU_STANDARD_HEIGHT.load(Ordering::Relaxed);
                    if std_width == 0 || std_height == 0 {
                        std_width = page_w_u32;
                        std_height = page_h_u32;
                    }
                    let standard_dims = StandardPageDimensions {
                        width: std_width,
                        height: std_height,
                    };
                    let target_height = if self.config.target_height() > 0 {
                        self.config.target_height()
                    } else {
                        standard_dims.height
                    };
                    let target_width = self
                        .config
                        .target_width()
                        .or_else(|| {
                            if target_height == standard_dims.height {
                                Some(standard_dims.width)
                            } else {
                                None
                            }
                        });

                    match margin::process_page_margins(
                        &adjusted_image,
                        &bounds,
                        margin_setting,
                        &standard_dims,
                        target_width,
                        target_height,
                    ) {
                        Ok(new_image) => {
                            let transformed = margin::transform_detections(
                                &detections,
                                &bounds,
                                margin_setting,
                                &standard_dims,
                                target_width,
                                target_height,
                                Some((page_w_u32, page_h_u32)),
                            );
                            dbglog!(
                                "[djvu-page] applied {:?} margins -> {}x{}",
                                margin_setting,
                                new_image.width(),
                                new_image.height()
                            );
                            detections = transformed;
                            adjusted_image = new_image;
                            page_w_u32 = adjusted_image.width();
                            page_h_u32 = adjusted_image.height();
                            page_w = page_w_u32 as usize;
                            page_h = page_h_u32 as usize;
                        }
                        Err(e) => {
                            eprintln!(
                                "Warning: Margin processing failed for page {}: {}",
                                page_index + 1,
                                e
                            );
                        }
                    }
                }
            }

            // Progress tracking for layout detection is now handled in batch processing

            // 4. Binarize the (possibly margin-adjusted) page image
            // Align inversion semantics with the main PDF path to avoid double-inversion:
            // - invert_input: pre-invert grayscale for dark-background docs (white text)
            // - invert (final): only apply if invert_input is not active
            let want_invert_input = self.config.invert_input();
            let mut want_invert_output = self.config.binarization.invert;
            if want_invert_input && want_invert_output {
                // Avoid double inversion which led to wrong polarity/all-black pages in DjVu
                want_invert_output = false;
                #[cfg(feature = "debug-logging")]
                crate::debug_log!(
                    "[DjVu] invert_input enabled; forcing final invert=false to avoid double inversion"
                );
            }

            let options = Legencode::types::BinarizationOptions {
                invert: want_invert_output,
                invert_input: want_invert_input,
                k_factor: self.config.binarization.k_factor,
                use_heavy_duty: self.config.binarization.use_heavy_duty
                    && !self.config.binarization.use_fixed_threshold,
                patch_percentage: self.config.binarization.patch_percentage,
                no_patch: self.config.binarization.no_patch,
                use_fixed_threshold: self.config.binarization.use_fixed_threshold,
                fixed_threshold: self.config.binarization.fixed_threshold,
            };

            // Use pooled binarized buffer to reduce allocations
            let binarized = with_djvu_binarized_buffer(page_w * page_h, |binarized_buffer| {
                // Prefer reusing heavy-duty processor when enabled and available
                let result = if options.use_heavy_duty {
                    HEAVY_SAUVOLA_PROCESSOR.with(|proc_cell| {
                        if let Some(proc_ref) = proc_cell.borrow_mut().as_mut() {
                            match apply_heavy_duty_binarization_raw(
                                proc_ref,
                                adjusted_image.as_raw(),
                                page_w,
                                page_h,
                                &options,
                            ) {
                                Ok(b) => b,
                                Err(e) => {
                                    #[cfg(feature = "debug-logging")]
                                    crate::debug_log!(
                                        "[djvu-page] Heavy-duty failed on page {}: {} — falling back to precision",
                                        page_index + 1,
                                        e
                                    );
                                    let mut fallback_opts = options.clone();
                                    fallback_opts.use_heavy_duty = false;
                                    Legencode::color::binarization::binarize_image_raw(
                                        adjusted_image.as_raw(),
                                        page_w,
                                        page_h,
                                        &fallback_opts,
                                    )
                                }
                            }
                        } else {
                            let mut fallback_opts = options.clone();
                            fallback_opts.use_heavy_duty = false;
                            Legencode::color::binarization::binarize_image_raw(
                                adjusted_image.as_raw(),
                                page_w,
                                page_h,
                                &fallback_opts,
                            )
                        }
                    })
                } else {
                    Legencode::color::binarization::binarize_image_raw(
                        adjusted_image.as_raw(),
                        page_w,
                        page_h,
                        &options,
                    )
                };
                 
                // Copy result to our buffer and return a clone
                binarized_buffer.clear();
                binarized_buffer.extend_from_slice(&result);
                result
            });

            // Note: We used to "whiten" (set to 255) pixels inside image regions here to avoid
            // encoding them in the JB2 layer. However, this is unnecessary because:
            // 1. The JB2 mask's white areas are inherently transparent
            // 2. The compose_color_background function correctly handles pre-masking
            // 3. The BG44 layer will show through the transparent regions regardless
            // Keeping the binarized mask as-is maintains better text detection quality.

            // 5. Update progress: page rendered/binarized
            if let Some(tracker) = &self.orchestrator_state.progress_tracker {
                let total = tracker.total_pages();
                // Increment rendered count and publish progress
                let rendered = self.orchestrator_state.rendered_count.fetch_add(1, Ordering::Relaxed) + 1;
                let detected = if !self.config.enable_layout_detection() {
                    // If layout detection is disabled, increment detected count too
                    self.orchestrator_state.detected_count.fetch_add(1, Ordering::Relaxed) + 1
                } else {
                    self.orchestrator_state.detected_count.load(Ordering::Relaxed)
                };
                let encoded = self.orchestrator_state.encoded_count.load(Ordering::Relaxed);
                tracker.publish_layout_progress(rendered as usize, detected as usize, encoded as usize, total);
            }

            // Determine if OCR should be skipped based on existing PDF text layer
            // DjVu behavior: if OCR is enabled, always regenerate OCR; otherwise no OCR
            let skip_ocr = false;

            // Create page data
            // Optionally run OCR (only if enabled in pipeline config)
            // Use region-based OCR when layout detection is enabled and we have detections,
            // otherwise use tiling-based OCR for robust coverage.
            let hocr_opt = if self.config.enable_ocr() {
                let use_regions = self.config.enable_layout_detection() && !detections.is_empty();
                let result = if use_regions {
                    crate::ocr::ocr::perform_region_based_ocr(
                        &binarized,
                        page_w,
                        page_h,
                        &detections,
                    )
                    .await
                } else {
                    crate::ocr::ocr::perform_tiling_based_ocr(&binarized, page_w, page_h).await
                };
                match result {
                    Ok(h) => if h.trim().is_empty() { None } else { Some(h) },
                        Err(e) => { dbglog!("[djvu-page] OCR failed on page {}: {}", page_index + 1, e); None }
                }
            } else {
                // OCR disabled: pass through existing PDF text layer if present
                if let Some(renderer) = env.get::<Arc<PdfiumRenderer>>("pdf_renderer") {
                    match renderer.has_text_layer(page_index as u32).await {
                        Ok(true) => {
                            if let Ok(raw_text) = renderer.extract_page_text(page_index as u32).await {
                                let s: String = raw_text;
                                let hocr = crate::build_hocr_from_pdf_text(&s, page_w as u32, page_h as u32);
                                if hocr.trim().is_empty() { None } else { Some(hocr) }
                            } else { None }
                        },
                        Ok(false) => None,
                        Err(e) => { dbglog!("[djvu-page] Text layer check failed on page {}: {}", page_index + 1, e); None }
                    }
                } else { None }
            };

            let page_data = PageData {
                index: page_index,
                rgb_image: adjusted_image.clone(),
                binarized,
                detections,
                hocr: hocr_opt,
            };

            // Process page through orchestrator
            let mut orchestrator = self.orchestrator_state.orchestrator.lock().await;
            if let Err(e) = orchestrator.process_page(page_data) {
                return Output::Err(format!("Failed to process page {}: {}", page_index, e));
            }
            drop(orchestrator);

            // Update progress: page encoded/exported
            if let Some(tracker) = &self.orchestrator_state.progress_tracker {
                let total = tracker.total_pages();
                // Increment encoded count and publish progress
                let encoded = self.orchestrator_state.encoded_count.fetch_add(1, Ordering::Relaxed) + 1;
                let rendered = self.orchestrator_state.rendered_count.load(Ordering::Relaxed);
                let detected = self.orchestrator_state.detected_count.load(Ordering::Relaxed);
                tracker.publish_layout_progress(rendered as usize, detected as usize, encoded as usize, total);
            }

            // Send completion signal
            let receivers = out_channels.get_receiver_ids();
            if !receivers.is_empty() {
            if let Err(_) = out_channels.send_to(&receivers[0], page_index).await {
                    dbglog!("[djvu-page] Failed to send completion signal for page {}", page_index);
                }
            }
            dbglog!("[djvu-page] finished page {}", page_index);
        }

        Output::Out(None)
    }
}

/// DJVU Document Assembler Action
/// Collects all processed pages and assembles final DJVU document
pub struct DjvuDocumentAssemblerAction {
    pub total_pages: usize,
    pub orchestrator_state: Arc<DjvuOrchestratorState>,
}

#[async_trait]
impl TypedAction for DjvuDocumentAssemblerAction {
    type I = usize; // Page index
    type O = (); // No output

    async fn run(
        &self,
        mut in_channels: TypedInChannels<Self::I>,
        _out_channels: TypedOutChannels<Self::O>,
        _env: Arc<EnvVar>,
    ) -> Output {
        let mut completed_pages = 0;
    #[cfg(feature = "djvu_debug")] dbglog!("[djvu-assembler] expecting {} pages", self.total_pages);

        // Notify UI/CLI we are starting final bundling immediately
        if let Some(tr) = &self.orchestrator_state.progress_tracker {
            tr.update(ProcessingStatus::AssemblingOutput);
        }

        // Collect all completed pages with their indices
        let mut completed_indices = Vec::with_capacity(self.total_pages);
        
        // Add timeout to prevent indefinite waiting
        let timeout_duration = std::time::Duration::from_secs(30); // 30 second timeout
        let start_time = std::time::Instant::now();
        
        loop {
            // Check for timeout
            if start_time.elapsed() > timeout_duration {
                dbglog!("[djvu-assembler] Timeout after 30s. Proceeding with {} of {} pages.", 
                         completed_indices.len(), self.total_pages);
                break;
            }
            
            // Try to receive with a shorter timeout to allow periodic timeout checks
            let recv_timeout = std::time::Duration::from_millis(500);
            match tokio::time::timeout(recv_timeout, in_channels.recv_any()).await {
                Ok(Ok((sender_id, page_index_opt))) => {
                    match page_index_opt {
                        Some(idx) => {
                            completed_indices.push(*idx);
                            let total_received = completed_indices.len();
                            
                            #[cfg(feature = "djvu_debug")] 
                            dbglog!(
                                "[djvu-assembler] received page_completion idx={} from_sender={:?} total_received={}", 
                                idx, sender_id, total_received
                            );
                            
                            if total_received >= self.total_pages { 
                                break; 
                            }
                        }
                        None => {
                            #[cfg(feature = "djvu_debug")] 
                            dbglog!("[djvu-assembler] channel closed by sender={:?} with None (break)", sender_id);
                            break;
                        }
                    }
                }
                Ok(Err(_)) => {
                    // Channel error - all senders likely closed
                    #[cfg(feature = "djvu_debug")] 
                    dbglog!("[djvu-assembler] all channels closed, proceeding with {} pages", completed_indices.len());
                    break;
                }
                Err(_) => {
                    // Timeout on recv - continue loop to check overall timeout
                    continue;
                }
            }
        }

        // Sort the completed pages to ensure correct order
        completed_indices.sort_unstable();
        let completed_pages = completed_indices.len();
        
        #[cfg(feature = "djvu_debug")] 
        dbglog!(
            "[djvu-assembler] loop end received={} expected={} complete_flag={}",
            completed_pages, self.total_pages, completed_pages == self.total_pages
        );

        // Check if we have any pages to work with
        if completed_pages == 0 {
            return Output::Err("No pages completed - cannot assemble DJVU document".to_string());
        }

        // Log missing pages but proceed with available ones
        if completed_pages != self.total_pages {
            let missing_pages: Vec<_> = (0..self.total_pages)
                .filter(|i| !completed_indices.contains(i))
                .collect();
            
            dbglog!(
                "[djvu-assembler] Only {} of {} pages completed. Missing pages: {:?}. Proceeding with available pages.",
                completed_pages, self.total_pages, missing_pages
            );
            
            // Log to help debug the issue
            dbglog!("[djvu-assembler] Available page indices: {:?}", completed_indices);
        }

        // Update the orchestrator's pages to be in the correct order
        {
            let mut orchestrator = self.orchestrator_state.orchestrator.lock().await;
            // Create a new vector with pages in the correct order
            let mut ordered_pages = Vec::with_capacity(self.total_pages);
            for &idx in &completed_indices {
                // Find the page with this index in the original pages
                if let Some(page) = orchestrator.pages.iter().find(|p| p.page_number == idx) {
                    ordered_pages.push(page.clone());
                } else {
                    return Output::Err(format!("Failed to find page {} in orchestrator", idx));
                }
            }
            // Replace the pages with the ordered version
            orchestrator.pages = ordered_pages;
        }

        // Write document metadata (for debugging/logging purposes)
        {
            let orchestrator = self.orchestrator_state.orchestrator.lock().await;
            if let Err(e) = orchestrator.write_document_metadata() { dbglog!("[djvu-assembler] Failed to write document metadata: {}", e); }
        }

        // Assemble DJVU document - now calls djvulibre executables directly
        let mut orchestrator = self.orchestrator_state.orchestrator.lock().await;
        let tracker_opt = self.orchestrator_state.progress_tracker.as_ref();
        dbglog!("[djvu-assembler] About to call assemble_djvu_with_progress; pages_stored={}", orchestrator.page_count());
        match orchestrator.assemble_djvu_with_progress(&self.orchestrator_state.output_path, tracker_opt) {
            Ok(()) => {
                println!(
                    "✓ DJVU document created: {:?}",
                    self.orchestrator_state.output_path
                );
                
                // Ensure terminal completion event is sent to unblock CLI
                if let Some(tr) = &self.orchestrator_state.progress_tracker {
                    tr.finish("DJVU document created successfully");
                }
                
                Output::Out(None)
            }
            Err(e) => {
                // Send terminal error event to unblock CLI
                if let Some(tr) = &self.orchestrator_state.progress_tracker {
                    tr.finish_with_error(e);
                }
                Output::Err("Failed to assemble DJVU document".to_string())
            }
        }
    }
}

/// Helper function to create DJVU pipeline configuration
pub fn create_djvu_pipeline_config(
    output_path: &std::path::Path,
    config: &PipelineConfig,
) -> Result<DjvuConfig> {
    // Determine work directory
    let work_dir = crate::app_dirs::djvu_work_dir_for(Some(output_path));

    let margin_setting = config.margin_settings();
    let center_active = matches!(margin_setting, crate::margin::MarginSettings::StandardizeAndCenter);
    let crop_active = matches!(margin_setting, crate::margin::MarginSettings::CropAndResize);

    // Create DJVU config
    let djvu_config = DjvuConfig {
        dpi: 300, // Could be made configurable
        clean: true,
        lossy: None, // Could be made configurable
        iw44_decibels: 32,  // Balanced quality for ebooks
        work_dir,
        // Enable early page assembly only when neither margin mode is active
        early_page_assembly: !(center_active || crop_active),
        // Use PBM pre-masking of color layer (align with Default)
        pre_mask_color_layer: true,
        center_margins: center_active,
        crop_margins: crop_active,
    };

    Ok(djvu_config)
}

/// Helper function to reuse grayscale buffer for RGB to grayscale conversion
fn with_djvu_grayscale_buffer<F, R>(min_size: usize, f: F) -> R
where
    F: FnOnce(&mut Vec<u8>) -> R,
{
    DJVU_GRAYSCALE_BUFFER.with(|buffer| {
        let mut buf = buffer.borrow_mut();
        if buf.len() < min_size {
            buf.resize(min_size, 0);
        }
        f(&mut buf)
    })
}

/// Helper function to reuse inference buffer for image preprocessing
fn with_djvu_inference_buffer<F, R>(min_size: usize, f: F) -> R
where
    F: FnOnce(&mut Vec<u8>) -> R,
{
    DJVU_INFERENCE_BUFFER.with(|buffer| {
        let mut buf = buffer.borrow_mut();
        if buf.len() < min_size {
            buf.resize(min_size, 0);
        }
        f(&mut buf)
    })
}

/// Helper function to create a reusable image buffer for high-res processing
fn with_reusable_image_buffer<F, R>(width: u32, height: u32, f: F) -> R
where
    F: FnOnce(&mut Vec<u8>) -> R,
{
    let required_size = (width * height * 3) as usize;
    with_djvu_temp_buffer(required_size, |buffer| {
        // Ensure buffer has correct size
        buffer.resize(required_size, 0);
        f(buffer)
    })
}
